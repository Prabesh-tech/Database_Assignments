1. SELECT Procedure (for STUDENT table)




CREATE OR REPLACE PROCEDURE proc_select_student
(
    vn_student_id STUDENT.student_id%TYPE
)
IS
    vc_student_name STUDENT.student_name%TYPE;
BEGIN
    SELECT student_name
    INTO vc_student_name
    FROM STUDENT
    WHERE student_id = vn_student_id;

    DBMS_OUTPUT.PUT_LINE('Student Name: ' || vc_student_name);

END proc_select_student;
