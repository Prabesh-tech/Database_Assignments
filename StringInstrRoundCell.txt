1. CEIL()

Rounds a number up to the nearest integer.

Syntax
SELECT CEIL(AVG(salary)) AS "Ceil Salary"
FROM staff;
Use Cases
Round average salary upward
SELECT CEIL(AVG(salary)) FROM staff;
Calculate required rooms
SELECT CEIL(125/10) AS rooms_needed FROM dual;
Round up product price
SELECT product_name, CEIL(price)
FROM products;
2. FLOOR()

Rounds a number down to nearest integer.

Syntax
SELECT FLOOR(AVG(salary))
FROM staff;
Use Cases
Lower integer salary value
SELECT FLOOR(AVG(salary)) FROM staff;
Whole discount calculation
SELECT FLOOR(price * 0.15)
FROM products;
Count full years
SELECT FLOOR(months_between(sysdate, hire_date)/12)
FROM employees;
3. ROUND()

Rounds number to specified decimal places.

Syntax
SELECT ROUND(AVG(salary),2)
FROM staff;
Use Cases
Display salary with 2 decimals
SELECT ROUND(AVG(salary),2)
FROM staff;
Round tax amount
SELECT ROUND(price * 0.13,2)
FROM products;
Round GPA
SELECT ROUND(gpa,1)
FROM students;
CHARACTER FUNCTIONS
4. CONCAT()

Joins two strings.

Syntax
SELECT CONCAT(first_name, last_name)
FROM people;
Use Cases
Create full name
SELECT CONCAT(first_name, last_name)
FROM people;
Generate username
SELECT CONCAT(first_name, '_')
FROM people;
Create message
SELECT CONCAT(surname, ' is manager')
FROM employees;
5. SUBSTR()

Extracts part of a string.

Syntax
SELECT SUBSTR('CAROLE',3,4)
FROM dual;

Result:

ROLE
Use Cases
Get first 3 letters
SELECT SUBSTR(firstname,1,3)
FROM staff;
Extract domain
SELECT SUBSTR(email, INSTR(email,'@')+1)
FROM users;
Create short codes
SELECT SUBSTR(product_name,1,5)
FROM products;
6. TRIM()

Removes unwanted characters/spaces.

Syntax
SELECT TRIM(' ' FROM firstname)
FROM staff;
Use Cases
Remove spaces
SELECT TRIM(firstname)
FROM staff;
Remove leading character
SELECT TRIM(LEADING '0' FROM phone)
FROM contacts;
Remove trailing character
SELECT TRIM(TRAILING '.' FROM sentence)
FROM notes;
7. LENGTH()

Returns number of characters.

Syntax
SELECT LENGTH('CAROLE')
FROM dual;
Use Cases
Check name length
SELECT LENGTH(firstname)
FROM staff;
Validate password size
SELECT username
FROM users
WHERE LENGTH(password) < 8;
Find longest surname
SELECT MAX(LENGTH(surname))
FROM staff;
8. INSTR()

Finds position of substring.

Syntax
SELECT INSTR('CORPORATE FLOOR', 'OR', 3, 2)
FROM dual;
Explanation
Start searching from position 3
Find the 2nd occurrence of 'OR'
Use Cases
Find @ in email
SELECT INSTR(email,'@')
FROM users;
Find second space
SELECT INSTR(fullname,' ',1,2)
FROM people;
Extract file extension
SELECT SUBSTR(filename, INSTR(filename,'.')+1)
FROM files;
