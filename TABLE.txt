1. PERSON TABLE

CREATE TABLE person (
    person_id           NUMBER          PRIMARY KEY,
    first_name          VARCHAR(50),
    middle_name         VARCHAR(50),
    last_name           VARCHAR(50),
    gender              VARCHAR(10),
    date_of_birth       DATE,
    age                 NUMBER,
    nationality         VARCHAR(50),
    marital_status      VARCHAR(20),
    blood_group         VARCHAR(5),
    religion            VARCHAR(50),
    language            VARCHAR(50)
);
