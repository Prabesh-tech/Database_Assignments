1. Procedure to insert into STUDENT table


CREATE OR REPLACE PROCEDURE proc_add_student
(
    vc_student_name STUDENT.student_name%TYPE
)
IS
BEGIN
    INSERT INTO STUDENT
    VALUES (seq_student_id.NEXTVAL, vc_student_name);

END proc_add_student;
